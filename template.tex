\begin{frame}[fragile]{A readable code snippet}
  \framesubtitle{Use a fragile frame for literal code; keep the example focused}

\small
\begin{Verbatim}
def normalise(values):
    """Scale a sequence to the range [0, 1]."""
    low, high = min(values), max(values)
    if high == low:
        return [0.0 for _ in values]
    return [(x - low) / (high - low)
            for x in values]
\end{Verbatim}
      \end{beamercolorbox}
    \column{.36\textwidth}
      \begin{block}{Explain the useful part}
        The constant-input case avoids division by zero. The remaining values share a common scale.
      \end{block}

      \par\medskip
      \textcolor{DASEslate}{\scriptsize Code lines do not use title truncation. Break long lines explicitly.}

\begin{frame}{What happens when an input is too long?}
  \framesubtitle{The next chapter deliberately uses oversized text to demonstrate the safeguards}
  \small
  \begin{tabular}{@{}p{.40\textwidth}p{.54\textwidth}@{}}
    \toprule
    \textbf{Theme placeholder} & \textbf{Automatic behaviour}\\
    \midrule
    Slide title / subtitle & One line, then an ellipsis.\\[2mm]
    Cover title / subtitle & At most two lines, then an ellipsis.\\[2mm]
    Chapter title & Up to eight full lines; ellipsis only if more follow. The number stays vertically centred.\\[2mm]
    Block heading / footer fields & One line, then an ellipsis.\\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{alertblock}{The original font size stays fixed}
    The full text remains in your source. The theme shortens only its visible presentation; shorter wording is still better for your audience.
  \end{alertblock}

\end{frame}
