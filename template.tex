\section{The visual language}

\begin{frame}{A small palette, a clear hierarchy}
  \framesubtitle{The crest supplies the energy; DASE purple keeps the system coherent}
  \begin{columns}[T,onlytextwidth]
    \column{.52\textwidth}
      \begin{tikzpicture}[x=1cm,y=1cm]
        \foreach \x/\c/\name in {0/DASEpurple/DASE purple,1.15/HKUred/HKU red,2.3/DASEgreen/DASE green,3.45/DASEblue/DASE blue,4.6/DASEyellow/DASE yellow} {
          \fill[\c,rounded corners=2pt] (\x,1.2) rectangle ++(1,1);
          \node[anchor=north west,font=\scriptsize\sffamily,text width=1.02cm] at (\x,1.08) {\name};
        }
      \end{tikzpicture}
      \vspace{0.95cm}
      \begin{itemize}
        \item Dark text on a warm white canvas keeps body copy comfortable.
        \item Purple marks structure: titles, section pages, and primary actions.
        \item Orange and yellow are reserved for emphasis, so they retain their signal.
      \end{itemize}
    \column{.42\textwidth}
      \begin{block}{Type scale}
        {\fontsize{22}{25}\selectfont\bfseries Slide title}\par\medskip
        {\large\bfseries A useful subtitle}\par\smallskip
        {\small Body copy should stay around 18--22pt on screen. Keep lines short and let the margins breathe.}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Research question $\;\to\;$ evidence $\;\to\;$ decision}
  \begin{center}
    \begin{tikzpicture}[node distance=7mm and 9mm,every node/.style={align=center},
      card/.style={draw=DASEline,rounded corners=3pt,minimum width=39mm,minimum height=25mm,inner sep=4mm},
      arrow/.style={-{Latex[length=3mm]},very thick,DASEpurple}]
      \node[card,fill=DASEpurple!10] (question) {\color{DASEpurple}\Large 01\\[-1mm]\normalsize Define the behaviour\\[1mm]\scriptsize What should change?};
      \node[card,fill=DASEgreen!11,right=of question] (evidence) {\color{DASEgreen!60!black}\Large 02\\[-1mm]\normalsize Build the test\\[1mm]\scriptsize What would count as evidence?};
      \node[card,fill=DASEorange!11,right=of evidence] (decision) {\color{DASEorange}\Large 03\\[-1mm]\normalsize Make the call\\[1mm]\scriptsize What can we claim?};
      \draw[arrow] (question)--(evidence);
      \draw[arrow] (evidence)--(decision);
    \end{tikzpicture}
  \end{center}
  \vfill
  \begin{alertblock}{Design principle}
    A slide is easier to trust when the claim, the measurement, and the consequence appear together.
  \end{alertblock}
\end{frame}
